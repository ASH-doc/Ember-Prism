cmake_minimum_required(VERSION 3.22)
project(EmberVA VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(EMBER_BUILD_PLUGIN "Build JUCE standalone and VST3" ON)
add_library(SynthACore Source/sa-parameters.cpp Source/sa-engine.cpp Source/sa-effects.cpp Source/sa-sequencer.cpp)
target_include_directories(SynthACore PUBLIC Source)
set_target_properties(SynthACore PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(EmberTests Tests/sa-tests.cpp)
target_link_libraries(EmberTests PRIVATE SynthACore)
enable_testing()
add_test(NAME SynthAEngine COMMAND EmberTests)
if(EMBER_BUILD_PLUGIN)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE/CMakeLists.txt")
    add_subdirectory(third_party/JUCE EXCLUDE_FROM_ALL)
  else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG f72bad64d29715216226685810c5196bd0d79d77 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(JUCE)
  endif()
  juce_add_plugin(EmberVA COMPANY_NAME "Synth A Studio" BUNDLE_ID "studio.syntha.emberva"
    PLUGIN_MANUFACTURER_CODE Syas PLUGIN_CODE Emva FORMATS VST3 Standalone
    PRODUCT_NAME "Ember VA" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE
    VST3_CATEGORIES Instrument Synth)
  juce_generate_juce_header(EmberVA)
  juce_add_binary_data(EmberBank SOURCES Presets/factory.json)
  target_sources(EmberVA PRIVATE Source/sa-processor.cpp Source/sa-editor.cpp Source/sa-patches.cpp)
  target_compile_definitions(EmberVA PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(EmberVA PRIVATE SynthACore EmberBank juce::juce_audio_utils juce::juce_dsp
    PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
  juce_add_console_app(EmberIntegration PRODUCT_NAME "Ember Integration")
  target_sources(EmberIntegration PRIVATE Tests/sa-integration.cpp)
  target_include_directories(EmberIntegration PRIVATE Source "${CMAKE_CURRENT_BINARY_DIR}/EmberVA_artefacts/JuceLibraryCode")
  target_link_libraries(EmberIntegration PRIVATE EmberVA EmberBank juce::juce_audio_utils)
  add_test(NAME SynthAIntegration COMMAND EmberIntegration)
  juce_add_console_app(EmberPluginProbe PRODUCT_NAME "Ember Plugin Probe")
  target_sources(EmberPluginProbe PRIVATE Tests/sa-plugin-probe.cpp)
  target_compile_definitions(EmberPluginProbe PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1)
  target_link_libraries(EmberPluginProbe PRIVATE juce::juce_audio_utils)
endif()
