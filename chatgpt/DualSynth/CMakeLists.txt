cmake_minimum_required(VERSION 3.22)
project(DualSynth VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 add_compile_options(/utf-8)
endif()
add_subdirectory(third_party/JUCE EXCLUDE_FROM_ALL)
juce_add_binary_data(EmberBank NAMESPACE EmberBank HEADER_NAME EmberBankData.h SOURCES Engines/A/Presets/factory.json Engines/A/Presets/signal_studies.json)
juce_add_binary_data(EmberClaudeBank NAMESPACE EmberClaudeBank HEADER_NAME EmberClaudeBankData.h SOURCES Engines/A/Presets/claude-import.json)
juce_add_binary_data(PrismBank NAMESPACE PrismBank HEADER_NAME PrismBankData.h SOURCES Engines/B/Presets/factory.json Engines/B/Presets/signal_studies.json Engines/B/Presets/claude-prism.json)
juce_add_plugin(DualSynth COMPANY_NAME "Dual Synth Studio" BUNDLE_ID "studio.dualsynth.instrument"
 PLUGIN_MANUFACTURER_CODE Dyst PLUGIN_CODE Du41 PRODUCT_NAME "AEther v4.1"
 FORMATS Standalone VST3 IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
 EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Instrument Synth)
juce_generate_juce_header(DualSynth)
file(GLOB engine_a CONFIGURE_DEPENDS Engines/A/Source/*.cpp)
file(GLOB engine_b CONFIGURE_DEPENDS Engines/B/Source/*.cpp)
target_sources(DualSynth PRIVATE ${engine_a} ${engine_b} Source/Host.cpp Source/Editor.cpp)
target_include_directories(DualSynth PUBLIC Source Engines/A/Source Engines/B/Source)
target_compile_definitions(DualSynth PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(DualSynth PRIVATE EmberBank PrismBank EmberClaudeBank juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
juce_add_console_app(DualTests PRODUCT_NAME "Dual Synth Tests")
target_sources(DualTests PRIVATE Tests/HostTests.cpp)
target_include_directories(DualTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/DualSynth_artefacts/JuceLibraryCode")
target_link_libraries(DualTests PRIVATE DualSynth EmberBank PrismBank EmberClaudeBank juce::juce_audio_utils juce::juce_dsp)
juce_add_console_app(DualCatalog PRODUCT_NAME "Dual Synth Parameter Catalog")
target_sources(DualCatalog PRIVATE Tests/CatalogDump.cpp)
target_include_directories(DualCatalog PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/DualSynth_artefacts/JuceLibraryCode")
target_link_libraries(DualCatalog PRIVATE DualSynth EmberBank PrismBank EmberClaudeBank juce::juce_audio_utils juce::juce_dsp)
juce_add_console_app(DualClaudeImport PRODUCT_NAME "Dual Synth Claude Import")
target_sources(DualClaudeImport PRIVATE Tools/ClaudeImport/ClaudeImport.cpp)
target_include_directories(DualClaudeImport PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/DualSynth_artefacts/JuceLibraryCode" Tools/ClaudeImport)
target_link_libraries(DualClaudeImport PRIVATE DualSynth EmberBank PrismBank EmberClaudeBank juce::juce_audio_utils juce::juce_dsp)
juce_add_console_app(DualProbe PRODUCT_NAME "Dual Synth VST3 Probe")
target_sources(DualProbe PRIVATE Tests/PluginProbe.cpp)
target_compile_definitions(DualProbe PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1)
target_link_libraries(DualProbe PRIVATE juce::juce_audio_utils)
juce_add_console_app(DualSourceHost PRODUCT_NAME "Dual Synth Source Host")
target_sources(DualSourceHost PRIVATE Tools/SourceHost/SourceHost.cpp)
target_compile_definitions(DualSourceHost PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1)
target_link_libraries(DualSourceHost PRIVATE juce::juce_audio_utils)
# Serum / Spire import (contract v1): decoders and shared types, no JUCE GUI or audio dependency.
add_library(AEtherImport STATIC Source/Import/SpireDecoder.cpp)
target_include_directories(AEtherImport PUBLIC Source)
target_link_libraries(AEtherImport PUBLIC juce::juce_core PRIVATE juce::juce_recommended_config_flags)
juce_add_console_app(DualImportTests PRODUCT_NAME "Dual Synth Import Tests")
target_sources(DualImportTests PRIVATE Tests/ImportTests.cpp)
target_link_libraries(DualImportTests PRIVATE AEtherImport juce::juce_core)
juce_add_console_app(DualCompareWav PRODUCT_NAME "Dual Synth Compare WAV")
target_sources(DualCompareWav PRIVATE Tools/CompareWav/CompareWav.cpp)
target_include_directories(DualCompareWav PRIVATE Tools/ClaudeImport)
target_link_libraries(DualCompareWav PRIVATE juce::juce_audio_formats)
juce_add_console_app(DualSurvey PRODUCT_NAME "Dual Synth Library Survey")
target_sources(DualSurvey PRIVATE Tools/SourceSurvey/Survey.cpp)
target_link_libraries(DualSurvey PRIVATE AEtherImport juce::juce_core)
enable_testing()
add_test(NAME HostIntegration COMMAND DualTests)
add_test(NAME ImportDecoders COMMAND DualImportTests)
get_target_property(vst_path DualSynth_VST3 JUCE_PLUGIN_ARTEFACT_FILE)
add_test(NAME ActualVST3 COMMAND DualProbe "${vst_path}")
