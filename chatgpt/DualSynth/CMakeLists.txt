cmake_minimum_required(VERSION 3.22)
project(DualSynth VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 add_compile_options(/utf-8)
endif()
add_subdirectory(third_party/JUCE EXCLUDE_FROM_ALL)
juce_add_binary_data(EmberBank NAMESPACE EmberBank HEADER_NAME EmberBankData.h SOURCES Engines/A/Presets/factory.json)
juce_add_binary_data(PrismBank NAMESPACE PrismBank HEADER_NAME PrismBankData.h SOURCES Engines/B/Presets/factory.json)
juce_add_plugin(DualSynth COMPANY_NAME "Dual Synth Studio" BUNDLE_ID "studio.dualsynth.instrument"
 PLUGIN_MANUFACTURER_CODE Dyst PLUGIN_CODE Du01 PRODUCT_NAME "Dual Synth"
 FORMATS Standalone VST3 IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
 EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Instrument Synth)
juce_generate_juce_header(DualSynth)
file(GLOB engine_a CONFIGURE_DEPENDS Engines/A/Source/*.cpp)
file(GLOB engine_b CONFIGURE_DEPENDS Engines/B/Source/*.cpp)
target_sources(DualSynth PRIVATE ${engine_a} ${engine_b} Source/Host.cpp Source/Editor.cpp)
target_include_directories(DualSynth PUBLIC Source Engines/A/Source Engines/B/Source)
target_compile_definitions(DualSynth PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(DualSynth PRIVATE EmberBank PrismBank juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
juce_add_console_app(DualTests PRODUCT_NAME "Dual Synth Tests")
target_sources(DualTests PRIVATE Tests/HostTests.cpp)
target_include_directories(DualTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/DualSynth_artefacts/JuceLibraryCode")
target_link_libraries(DualTests PRIVATE DualSynth EmberBank PrismBank juce::juce_audio_utils juce::juce_dsp)
juce_add_console_app(DualProbe PRODUCT_NAME "Dual Synth VST3 Probe")
target_sources(DualProbe PRIVATE Tests/PluginProbe.cpp)
target_compile_definitions(DualProbe PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1)
target_link_libraries(DualProbe PRIVATE juce::juce_audio_utils)
enable_testing()
add_test(NAME HostIntegration COMMAND DualTests)
get_target_property(vst_path DualSynth_VST3 JUCE_PLUGIN_ARTEFACT_FILE)
add_test(NAME ActualVST3 COMMAND DualProbe "${vst_path}")
