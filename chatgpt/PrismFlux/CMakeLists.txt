cmake_minimum_required(VERSION 3.22)
project(PrismFlux VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(SynthBEngine Source/sb-schema.cpp Source/sb-tables.cpp Source/sb-filters.cpp Source/sb-motion.cpp Source/sb-rack.cpp Source/sb-engine.cpp)
target_include_directories(SynthBEngine PUBLIC Source)
set_target_properties(SynthBEngine PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(PrismCoreTests Tests/sb-core-tests.cpp)
target_link_libraries(PrismCoreTests PRIVATE SynthBEngine)
enable_testing()
add_test(NAME Core COMMAND PrismCoreTests)
option(PRISM_NATIVE "Build native instrument formats" ON)
if(PRISM_NATIVE)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE/CMakeLists.txt")
   add_subdirectory(third_party/JUCE EXCLUDE_FROM_ALL)
 else()
   include(FetchContent)
   FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG f72bad64d29715216226685810c5196bd0d79d77)
   FetchContent_MakeAvailable(JUCE)
 endif()
 juce_add_plugin(PrismFlux COMPANY_NAME "Synth B Studio" BUNDLE_ID "studio.synthb.prismflux"
   PLUGIN_MANUFACTURER_CODE Sybs PLUGIN_CODE Prfx PRODUCT_NAME "Prism Flux"
   FORMATS Standalone VST3 IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
   EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Instrument Synth)
 juce_generate_juce_header(PrismFlux)
 juce_add_binary_data(PrismFactory SOURCES Presets/factory.json)
 target_sources(PrismFlux PRIVATE Source/sb-plugin.cpp Source/sb-patches.cpp Source/sb-ui.cpp Source/sb-ui-motion.cpp)
 target_compile_definitions(PrismFlux PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(PrismFlux PRIVATE SynthBEngine PrismFactory juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
 juce_add_console_app(PrismNativeTests PRODUCT_NAME "Prism Native Tests")
 target_sources(PrismNativeTests PRIVATE Tests/sb-native-tests.cpp)
 target_include_directories(PrismNativeTests PRIVATE Source "${CMAKE_CURRENT_BINARY_DIR}/PrismFlux_artefacts/JuceLibraryCode")
 target_link_libraries(PrismNativeTests PRIVATE PrismFlux PrismFactory juce::juce_audio_utils)
 add_test(NAME Native COMMAND PrismNativeTests)
 juce_add_console_app(PrismVST3Check PRODUCT_NAME "Prism VST3 Check")
 target_sources(PrismVST3Check PRIVATE Tests/sb-vst3-check.cpp)
 target_compile_definitions(PrismVST3Check PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1)
 target_link_libraries(PrismVST3Check PRIVATE juce::juce_audio_utils)
 get_target_property(prism_vst3_bundle PrismFlux_VST3 JUCE_PLUGIN_ARTEFACT_FILE)
 add_test(NAME VST3Binary COMMAND PrismVST3Check "${prism_vst3_bundle}")
endif()
